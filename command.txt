# ==========================================
# Python & Selenium Installation Commands
# ==========================================

# 1. Verify Python & pip are installed (Python 3.10+ recommended)
python --version
pip --version

# (Optional: Install Python via Windows Package Manager if not installed)
# winget install Python.Python.3.12

# 2. Upgrade pip (recommended)
python -m pip install --upgrade pip

# 3. Install Selenium and test dependencies
pip install selenium pytest webdriver-manager
